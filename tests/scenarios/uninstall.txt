# uninst.exe без параметров (двойной щелчок): экран удаления, отмена.
run "%XPS_TARGET%\uninst.exe" /log "%XPS_LOG%"
expect "подготовлена к удалению" 15000
key ESC
sleep 3000
# Удаление с параметром, как из «Программ и компонентов».
run "%XPS_TARGET%\uninst.exe" /uninstall /log "%XPS_LOG%"
expect "подготовлена к удалению" 15000
# На экране ещё может быть прежний текст: ждать запуска новой копии.
sleep 2500
key L
expect "удалена с этого компьютера" 30000
key ENTER
sleep 5000
